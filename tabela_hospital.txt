CREATE DATABASE DB_HOSPITAL;
USE DB_HOSPITAL;

CREATE TABLE MAE(
    ID_MAE INT NOT NULL,
    NOME_MAE VARCHAR(100) NOT NULL,
    ENDERECO VARCHAR(100) NOT NULL,
    TELEFONE VARCHAR(100) NOT NULL,
    DATA_DE_ANIVERSARIO DATE,
    CONSTRAINT PK_MAE PRIMARY KEY(ID_MAE)
);

CREATE TABLE BEBES(
    ID INT NOT NULL,
    NOME_BEBE VARCHAR(100) NOT NULL,
    DATA_NASC DATE NOT NULL,
    PESO_AO_NASCER numeric NOT NULL, 
    TAMANHO numeric NOT NULL,       
    ID_MAE INT NOT NULL,
    CONSTRAINT PK_BEBES PRIMARY KEY(ID),
    CONSTRAINT FK_BEBES_MAE FOREIGN KEY (ID_MAE) REFERENCES MAE (ID_MAE)
);

CREATE TABLE DOUTOR(
    ID_DOCTOR INT NOT NULL,
    CRM VARCHAR(6) NOT NULL,
    NOME VARCHAR(100) NOT NULL,
    TELEFONE VARCHAR(100) NOT NULL,
    ESPECIALIDADE VARCHAR(200) NOT NULL,
    CONSTRAINT PK_DOUTOR PRIMARY KEY(ID_DOCTOR)
);

CREATE TABLE Consulta(
    ID_CONSULTA INT NOT NULL, 
    ID_BEBE INT NOT NULL,
    ID_DOUTOR INT NOT NULL,
    DATA_SERVICO DATE NOT NULL,
    TEMPO_SERVICO TIME NOT NULL,
    CONSTRAINT PK_CONSULTA PRIMARY KEY(ID_CONSULTA),
    CONSTRAINT FK_CONSULTA_BEBE FOREIGN KEY (ID_BEBE) REFERENCES BEBES (ID), 
    CONSTRAINT FK_CONSULTA_DOUTOR FOREIGN KEY (ID_DOUTOR) REFERENCES DOUTOR (ID_DOCTOR)
);
